## Context

IgBLAST requires a combined VDJC file named `{species}_V.fasta` in `internal_data/` for correct `complete_vdj` calculation. Currently, `build_internal_data.py` creates symlinks to separate V/D/J/C files in `database/`, causing IgBLAST to miscalculate `complete_vdj` for all species.

**Current state (Broken):**
- `build_internal_data.py` creates symlinks: `human_V.* -> ../../../database/human/human_V.*`
- Separate files: `human_V.*`, `human_D.*`, `human_J.*`, `human_C.*`
- Result: IgBLAST cannot find D/J/C segments in `_V` file → `complete_vdj` fails

**G3 Reference (Working):**
- Single `human_V.fasta` containing ALL segments (479 sequences: V+D+J+C)
- BLAST database built from combined file
- No separate `_D`, `_J`, `_C` files in internal_data

**Goal:**
- Concatenate V+D+J+C FASTAs → single `{species}_V.fasta`
- Build BLAST database from combined file (actual files, not symlinks)
- Remove all symlinks from internal_data directories
- Update GermlineData to use `database/{species}/` for V/D/J/C BLAST searches

```python
import subprocess
from pathlib import Path

def build_combined_fasta(species: str, database_dir: Path, internal_data_dir: Path) -> Path:
    """
    Concatenate V+D+J+C FASTAs into single combined file.
    
    Parameters
    ----------
    species : str
        Species name
    database_dir : Path
        Path to database/{species}/ containing segment FASTAs
    internal_data_dir : Path
        Path to internal_data/{species}/ for output
        
    Returns
    -------
    Path
        Path to combined FASTA file
    """
    combined_fasta = internal_data_dir / f"{species}_V.fasta"
    
    # Ordered segments: V, D, J, C (D may not exist for all species)
    segments = ["V", "D", "J", "C"]
    
    with open(combined_fasta, "w") as out:
        for segment in segments:
            fasta_file = database_dir / f"{species}_{segment}.fasta"
            if fasta_file.exists():
                with open(fasta_file) as f:
                    content = f.read()
                    # Ensure file ends with newline before next segment
                    if content and not content.endswith("\n"):
                        content += "\n"
                    out.write(content)
                logger.info(f"Added {segment} genes from {fasta_file}")
            else:
                logger.debug(f"No {segment} genes found at {fasta_file}")
    
    return combined_fasta


def build_blast_db(fasta_path: Path, db_prefix: Path) -> bool:
    """
    Build BLAST database from FASTA file.
    
    Parameters
    ----------
    fasta_path : Path
        Path to input FASTA file
    db_prefix : Path
        Output database prefix (e.g., internal_data/human/human_V)
        
    Returns
    -------
    bool
        True if successful
    """
    cmd = [
        "makeblastdb",
        "-in", str(fasta_path),
        "-dbtype", "nucl",
        "-parse_seqids",
        "-out", str(db_prefix),
    ]
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, check=True)
        logger.info(f"Built BLAST database at {db_prefix}")
        return True
    except subprocess.CalledProcessError as e:
        logger.error(f"makeblastdb failed: {e.stderr}")
        return False
    except FileNotFoundError:
        logger.error("makeblastdb not found - ensure BLAST+ is installed")
        return False
```

```python
def build_internal_data(species: str, germlines_root: Path) -> bool:
    """Build internal_data directory for a species."""
    database_dir = germlines_root / "igblast" / "database" / species
    internal_data_dir = germlines_root / "igblast" / "Ig" / "internal_data" / species

    if not database_dir.exists():
        logger.error(f"Database directory not found: {database_dir}")
        return False

    # Create clean internal_data directory (remove old symlinks)
    if internal_data_dir.exists():
        for item in internal_data_dir.iterdir():
            if item.is_symlink():
                item.unlink()
    internal_data_dir.mkdir(parents=True, exist_ok=True)
    logger.info(f"Created: {internal_data_dir}")

    # Create combined VDJC FASTA
    combined_fasta = build_combined_fasta(species, database_dir, internal_data_dir)
    if not combined_fasta.exists() or combined_fasta.stat().st_size == 0:
        logger.error(f"Failed to create combined FASTA for {species}")
        return False
    
    # Build BLAST database from combined file
    db_prefix = internal_data_dir / f"{species}_V"
    if not build_blast_db(combined_fasta, db_prefix):
        return False

    # Generate NDM file (unchanged)
    ndm_entries = build_ndm_file(species, germlines_root)
    if not ndm_entries:
        logger.warning(f"No NDM entries generated for {species}")
        return False

    ndm_path = internal_data_dir / f"{species}.ndm.imgt"
    with open(ndm_path, "w") as f:
        f.write("\n".join(ndm_entries) + "\n")
    logger.info(f"Generated {ndm_path} with {len(ndm_entries)} entries")

    return True
```

```python
def _make_internal_annotaion_file(self, outpath: Path) -> None:
    """Generate the internal database file for IgBlast.
    
    Creates combined VDJC FASTA and BLAST database for each reference name.
    """
    logger.debug(f"Generating internal annotation file at {outpath}")
    
    database = self.get_dataframe()
    for name, group_df in database.groupby("name"):
        # V genes for NDM file (framework/CDR regions)
        v_genes = group_df.loc[group_df["gene_segment"] == "V"].copy()
        
        # All segments for combined FASTA
        all_segments = group_df.loc[group_df["gene_segment"].isin(["V", "D", "J", "C"])].copy()
        
        name_internal_df_path = Path(outpath).joinpath(Path(f"Ig/internal_data/{name}/"))
        if not name_internal_df_path.exists():
            logger.info(f"Creating {name_internal_df_path}")
            name_internal_df_path.mkdir(parents=True)

        # Generate NDM file from V genes only (unchanged)
        if not v_genes.empty:
            index_df = v_genes[
                [
                    "gene",
                    "imgt.fwr1_start",
                    "imgt.fwr1_end",
                    "imgt.cdr1_start",
                    "imgt.cdr1_end",
                    "imgt.fwr2_start",
                    "imgt.fwr2_end",
                    "imgt.cdr2_start",
                    "imgt.cdr2_end",
                    "imgt.fwr3_start",
                    "imgt.fwr3_end",
                ]
            ].copy()
            
            index_df = (index_df.set_index("gene") + 1).astype("Int64").reset_index()
            index_df = index_df.drop(index_df[index_df.isna().any(axis=1)].index)
            
            scheme = "imgt"
            internal_annotations_file_path = name_internal_df_path.joinpath(f"{name}.ndm.{scheme}")
            
            segment = [i.split("|")[-1].split("-")[0][0:4][::-1][:2] for i in index_df["gene"]]
            index_df["segment"] = segment
            index_df["weird_buffer"] = 0
            logger.info(f"Writing to annotation file {internal_annotations_file_path}")
            index_df.to_csv(internal_annotations_file_path, sep="\t", header=False, index=False)

        # Build combined VDJC BLAST database
        suffix = "V"
        db_outpath = Path(str(name_internal_df_path) + f"/{name}_{suffix}")
        
        # Use ALL segments for BLAST database, not just V genes
        make_blast_db_for_internal(all_segments, db_outpath)
        logger.info(f"Built combined VDJC database with {len(all_segments)} sequences")
```

Note: The `make_blast_db_for_internal()` function already handles writing FASTA and building BLAST db. Just need to pass all segments instead of filtered V-only data.

```python
elif _use_germlines_module():
    germlines_igblast = _get_germlines_igblast_dir()
    internal_data_species = germlines_igblast / "Ig" / "internal_data" / name
    database_species = germlines_igblast / "database" / name

    if internal_data_species.exists():
        self.base_dir = germlines_igblast
        
        # V/D/J/C BLAST databases are in database/{species}/
        # These are used by IgBLAST for -germline_db_V, -germline_db_D, etc.
        blast_prefix = database_species / f"{name}_"
        self.blast_dir = blast_prefix
        self.v_gene_dir = Path(str(blast_prefix) + "V")
        self.d_gene_dir = Path(str(blast_prefix) + "D")
        self.j_gene_dir = Path(str(blast_prefix) + "J")
        self.c_gene_dir = Path(str(blast_prefix) + "C")
        
        # IGDATA points to Ig/ which contains internal_data/{species}/
        # internal_data has combined VDJC file named {species}_V for annotation
        self.igdata = germlines_igblast / "Ig"
        
        self.aux_path = germlines_igblast / "aux_db" / f"{name}_gl.aux"
```

**Important notes:**
1. `igdata` must point to `Ig/` directory - IgBLAST uses `IGDATA` env var to find `internal_data/{species}/{species}_V` for its internal annotation database
2. `v_gene_dir`, `d_gene_dir`, `j_gene_dir`, `c_gene_dir` must point to `database/{species}/` - these are passed to IgBLAST's `-germline_db_V`, `-germline_db_D`, `-germline_db_J`, `-germline_db_C` options for the actual alignment search
3. The combined file in `internal_data/` is ONLY for IgBLAST's internal annotation (complete_vdj calculation) - NOT for the germline alignment search
